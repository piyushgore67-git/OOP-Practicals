Assignment No. 3: Employee Bio-Data Management System (Multiple Inheritance & Virtual Base Class)

#include <iostream>
#include <string>
using namespace std;

// Virtual Base Class
class Employee
{
protected:
    int empId;
    string empName;

public:
    void acceptEmployee()
    {
        cout << "Enter Employee ID: ";
        cin >> empId;
        cin.ignore();
        cout << "Enter Employee Name: ";
        getline(cin, empName);
    }

    void displayEmployee()
    {
        cout << "\nEmployee ID        : " << empId;
        cout << "\nEmployee Name      : " << empName;
    }
};

// Derived Class 1 (Virtual Inheritance)
class PersonalRecord : virtual public Employee
{
protected:
    string dob;
    string address;
    string contactNo;

public:
    void acceptPersonal()
    {
        cout << "Enter Date of Birth (DD/MM/YYYY): ";
        getline(cin, dob);
        cout << "Enter Address: ";
        getline(cin, address);
        cout << "Enter Contact Number: ";
        getline(cin, contactNo);
    }

    void displayPersonal()
    {
        cout << "\nDate of Birth      : " << dob;
        cout << "\nAddress            : " << address;
        cout << "\nContact Number     : " << contactNo;
    }
};

// Derived Class 2 (Virtual Inheritance)
class ProfessionalRecord : virtual public Employee
{
protected:
    string designation;
    string department;
    string organization;
    int experience;

public:
    void acceptProfessional()
    {
        cout << "Enter Designation: ";
        getline(cin, designation);
        cout << "Enter Department: ";
        getline(cin, department);
        cout << "Enter Organization: ";
        getline(cin, organization);
        cout << "Enter Years of Experience: ";
        cin >> experience;
        cin.ignore();
    }

    void displayProfessional()
    {
        cout << "\nDesignation        : " << designation;
        cout << "\nDepartment         : " << department;
        cout << "\nOrganization       : " << organization;
        cout << "\nExperience         : " << experience << " years";
    }
};

// Derived Class 3 (Virtual Inheritance)
class AcademicRecord : virtual public Employee
{
protected:
    string qualification;
    string specialization;
    string university;
    int passingYear;

public:
    void acceptAcademic()
    {
        cout << "Enter Highest Qualification: ";
        getline(cin, qualification);
        cout << "Enter Specialization: ";
        getline(cin, specialization);
        cout << "Enter University: ";
        getline(cin, university);
        cout << "Enter Passing Year: ";
        cin >> passingYear;
    }

    void displayAcademic()
    {
        cout << "\nQualification      : " << qualification;
        cout << "\nSpecialization     : " << specialization;
        cout << "\nUniversity         : " << university;
        cout << "\nPassing Year       : " << passingYear;
    }
};

// Multiple Inheritance from all three record classes
class EmployeeBioData : public PersonalRecord, public ProfessionalRecord, public AcademicRecord
{
public:
    void accept()
    {
        cout << "\n--- Enter Employee Details ---\n";
        acceptEmployee();
        acceptPersonal();
        acceptProfessional();
        acceptAcademic();
    }

    void display()
    {
        cout << "\n========== EMPLOYEE BIO-DATA ==========";
        displayEmployee();
        cout << "\n\n--- Personal Details ---";
        displayPersonal();
        cout << "\n\n--- Professional Details ---";
        displayProfessional();
        cout << "\n\n--- Academic Details ---";
        displayAcademic();
        cout << "\n=======================================\n";
    }
};

int main()
{
    EmployeeBioData emp;
    emp.accept();
    emp.display();
    return 0;
}
