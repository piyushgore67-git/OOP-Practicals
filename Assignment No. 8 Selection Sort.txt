Assignment No. 8: Selection Sort Using Function Template

#include <iostream>
using namespace std;

// Function Template for Selection Sort
template <class T>
void sortArray(T arr[], int size)
{
    for (int i = 0; i < size - 1; i++)
    {
        int minIndex = i;
        for (int j = i + 1; j < size; j++)
        {
            if (arr[j] < arr[minIndex])
            {
                minIndex = j;
            }
        }
        // Swap smallest element with current element
        T temp = arr[i];
        arr[i] = arr[minIndex];
        arr[minIndex] = temp;
    }

    cout << "Sorted array: ";
    for (int i = 0; i < size; i++)
    {
        cout << arr[i] << " ";
    }
    cout << endl;
}

int main()
{
    int intArr[] = {45, 12, 78, 23, 9};
    int intSize = sizeof(intArr) / sizeof(intArr[0]);

    float floatArr[] = {4.5, 1.2, 7.8, 2.3, 0.9};
    int floatSize = sizeof(floatArr) / sizeof(floatArr[0]);

    cout << "===== SELECTION SORT USING FUNCTION TEMPLATE =====\n";

    cout << "\nInteger array:\n";
    sortArray(intArr, intSize);

    cout << "\nFloating-point array:\n";
    sortArray(floatArr, floatSize);

    return 0;
}
